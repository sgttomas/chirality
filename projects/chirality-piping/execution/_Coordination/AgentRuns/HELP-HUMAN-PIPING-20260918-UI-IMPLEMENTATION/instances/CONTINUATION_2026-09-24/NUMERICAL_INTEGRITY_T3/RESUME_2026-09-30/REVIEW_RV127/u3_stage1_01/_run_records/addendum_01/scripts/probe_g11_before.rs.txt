    // RV127 probe only (never committed): G11 at B. For each of the four user stiffness
    // values, the demo with its legacy pressures removed and that value removed from C-150.
    #[test]
    fn rv127_probe_g11_before_fix() {
        for missing in ["lateral", "axial", "angular", "torsional"] {
            let mut input = request_with_refused_joint();
            for case in &mut input.model.load_cases {
                case.primitive_loads.retain(|load| load.category != "pressure");
            }
            let joint = input.model.components.iter_mut().find(|c| c.id == "component:C-150").unwrap();
            let modifiers = joint.modifiers.as_mut().unwrap();
            match missing {
                "lateral" => modifiers.lateral_stiffness_user_value = None,
                "axial" => modifiers.axial_stiffness_user_value = None,
                "angular" => modifiers.angular_stiffness_user_value = None,
                _ => modifiers.torsional_stiffness_user_value = None,
            }
            let output = run_linear_static_preview(input);
            let consumed = output.results.iter().filter(|r| r.kind == "component_user_stiffness_macro_element_review").count();
            let blocking: Vec<&str> = output.diagnostics.iter().filter(|d| d.severity == "blocking").map(|d| d.code.as_str()).collect();
            eprintln!("RV127_G11_BEFORE missing={missing} status={} consumed_rows={consumed} results={} blocking={blocking:?}",
                output.status.mechanics, output.results.len());
        }
    }
