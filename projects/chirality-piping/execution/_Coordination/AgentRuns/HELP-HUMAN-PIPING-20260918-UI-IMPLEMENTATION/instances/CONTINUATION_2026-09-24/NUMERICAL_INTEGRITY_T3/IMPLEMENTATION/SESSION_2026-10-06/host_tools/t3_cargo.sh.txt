#!/bin/bash
# T3 host rule: one cargo job at a time across all T3 work (ROOT, 2026-10-05).
# Usage: WT/tools/t3_cargo.sh <cargo args...>   (blocks until the host-wide lock is free)
# Refuses to start if the memory guard is not running.
WT="${T3_WT:?set T3_WT to the T3 root, WT}"
if ! pgrep -f "$WT/guard/memguard.sh" >/dev/null; then
  echo "t3_cargo: memory guard not running; refusing to start cargo" >&2; exit 97
fi
LOG=$WT/guard/cargo_jobs.log
echo "$(date -u '+%FT%TZ') WAIT pid=$$ cwd=$PWD cargo $*" >> "$LOG"
exec /usr/bin/lockf -k "$WT/guard/cargo_job.lock" /bin/bash -c '
  echo "$(date -u "+%FT%TZ") START pid=$$ cwd=$PWD cargo $*" >> "'"$LOG"'"
  cargo "$@"; rc=$?
  echo "$(date -u "+%FT%TZ") END rc=$rc pid=$$ cwd=$PWD" >> "'"$LOG"'"
  exit $rc' t3_cargo "$@"
