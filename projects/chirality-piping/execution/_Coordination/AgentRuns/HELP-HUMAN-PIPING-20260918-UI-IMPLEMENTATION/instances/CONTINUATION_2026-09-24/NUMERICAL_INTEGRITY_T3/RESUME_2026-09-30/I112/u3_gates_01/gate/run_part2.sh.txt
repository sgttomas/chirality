#!/bin/bash
# I112 U3 both-entry gate part 2: the four dense 1,000-member runs, base B (ba500defa4) and candidate C (8a12de28db)
# interleaved, one at a time, TIMEOUT_LARGE 1800 s, with I13's gate_part2.py (F1b gate 2, e43c662b...) unchanged.
# Run ONLY as WT/tools/t3_exclusive.sh /bin/bash run_part2.sh <execution-dir>, one execution at a time.
# Derived from I61's fixed run_part2.sh (the v2 watcher). Differences: paths (WT/scratch/i112_u3/gate, WT/targets/i112-u3-gate);
# the interpreter WT/venv/bin/python (python 3.13.14); foreign_jobs.sh as adapted for the T3 lock system (see its header);
# the watcher's subshell is stopped by its own PID as I61's was.
# Before starting, wait until no foreign cargo/solver/test job has been seen for QUIET_POLLS (default 8) consecutive 15 s polls;
# during the run, a watcher records foreign jobs every 5 s.
set -u
WT=WT; G=$WT/scratch/i112_u3/gate; PY=$WT/venv/bin/python
OUT=$G/${1:-part2}
pgrep -f memguard.sh >/dev/null || { echo "MEMGUARD NOT RUNNING"; exit 9; }
quiet=0; echo "== waiting for a quiet host $(date -u +%FT%TZ)"
while [ $quiet -lt ${QUIET_POLLS:-8} ]; do n=$($G/foreign_jobs.sh | wc -l); if [ $n -eq 0 ]; then quiet=$((quiet+1)); else quiet=0; fi; sleep 15; done
echo "== quiet since $(date -u +%FT%TZ)"
mkdir -p $OUT
( while true; do n=$($G/foreign_jobs.sh | wc -l | tr -d " "); j=$($G/foreign_jobs.sh | head -3 | cut -c1-120 | tr "\t\n" "  "); printf "%s\t%s\t%s\t%s\n" "$(date -u +%FT%TZ)" "$n" "$(sysctl -n vm.loadavg)" "$j"; sleep 5; done ) > $OUT/host_watch.tsv 2>/dev/null &
W=$!
cd $G && $PY -B scripts/gate_part2.py scripts $WT/targets/i112-u3-gate/base/release/t3_p1_probe $WT/targets/i112-u3-gate/cand/release/t3_p1_probe gen_out $OUT > $OUT/driver.log 2>&1; echo "== part2 exit=$? $(date -u +%FT%TZ)"
kill $W
echo "foreign-job samples during the run: $(awk -F"\t" "\$2>0" $OUT/host_watch.tsv | wc -l | tr -d " ") of $(wc -l < $OUT/host_watch.tsv | tr -d " ")"
echo PART2-DONE $(date -u +%FT%TZ)
