#!/bin/bash
# I114 shim: every cargo goes through the T3 wrapper, with --locked --offline on every subcommand.
args=("$@")
if [ $# -gt 0 ] && [[ "$1" != -* ]]; then
  have_locked=0; have_offline=0
  for a in "$@"; do [ "$a" = "--locked" ] && have_locked=1; [ "$a" = "--offline" ] && have_offline=1; [ "$a" = "--" ] && break; done
  extra=(); [ $have_locked = 0 ] && extra+=(--locked); [ $have_offline = 0 ] && extra+=(--offline)
  # insert after the subcommand, before any "--"
  args=("$1" "${extra[@]}" "${@:2}")
fi
PATH="$I114_ORIG_PATH" exec WT/tools/t3_cargo.sh "${args[@]}"
