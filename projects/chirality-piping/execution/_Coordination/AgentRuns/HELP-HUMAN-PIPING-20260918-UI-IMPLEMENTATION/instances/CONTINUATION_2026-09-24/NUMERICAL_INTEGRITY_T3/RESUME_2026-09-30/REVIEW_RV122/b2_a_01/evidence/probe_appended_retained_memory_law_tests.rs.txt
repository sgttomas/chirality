/// RV122 probe B: on branch E a case's `pressure_regions` is `Some(vec)` with length 0; its
/// capacity is a typed heap owner that no D1.9 row reads (T03: actual capacities). The sections
/// owner, empty with spare capacity, is the control.
#[test]
fn rv122_probe_exact_regions_capacity_is_unread() {
    let report = admitted_typed(exact3(milestone()), |r| r.model.load_cases[0].pressure_regions = Some(Vec::with_capacity(1 << 16)));
    println!("RV122_REGIONS_CAPACITY domain={:?}", report.law().domain);
    let control = admitted_typed(exact3(milestone()), |r| r.model.sections.reserve_exact(1));
    println!("RV122_SECTIONS_CAPACITY domain={:?}", control.law().domain);
    assert!(control.law().domain.is_some(), "the control refuses");
}
