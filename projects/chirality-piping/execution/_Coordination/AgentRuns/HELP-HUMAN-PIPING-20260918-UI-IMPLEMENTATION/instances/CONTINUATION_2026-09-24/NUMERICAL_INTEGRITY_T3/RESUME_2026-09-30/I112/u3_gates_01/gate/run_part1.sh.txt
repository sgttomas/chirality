#!/bin/bash
# I112 U3 both-entry gate part 1: 884 runs per side with G1's gate_run_base_full.py (8cf952ae...), run.py and compare.py unchanged;
# K-D5's gate_check.py with the empty S11/FORMATION exception lists; KF2's compare_gate_kf2.py. Base first, then candidate.
# Derived from I61's run_part1.sh. Differences: paths (WT/scratch/i112_u3/gate, WT/targets/i112-u3-gate); the interpreter is
# WT/venv/bin/python (python 3.13.14; the REPO_ROOT .venv I61 named is not present on this host); each side's driver runs as
# one WT/tools/t3_slot.sh job (host rule); a diff -rq line for full/ and envelopes/ is kept as I61 did.
set -u
WT=WT; G=$WT/scratch/i112_u3/gate; PY=$WT/venv/bin/python
pgrep -f memguard.sh >/dev/null || { echo "MEMGUARD NOT RUNNING"; exit 9; }
cd $G
for n in base cand; do
  echo "== part1 $n start $(date -u +%FT%TZ)"
  $WT/tools/t3_slot.sh $PY -B scripts/gate_run_base_full.py $WT/targets/i112-u3-gate/$n/release/t3_p1_probe gen_out part1_$n 4 > logs/part1_${n}_driver.log 2>&1; echo "== part1 $n exit=$? $(date -u +%FT%TZ)"
  $PY -B scripts/gate_check.py gen_out part1_$n/runs.jsonl inputs/references.json part1_$n/result_part1_$n.json inputs/S11_EXCEPTIONS.json inputs/FORMATION_EXCEPTIONS.json > logs/part1_${n}_gate_check.log 2>&1; echo "== gate_check $n exit=$?"
done
$PY -B scripts/compare_gate_kf2.py part1_base part1_cand scripts part1_base/result_part1_base.json part1_cand/result_part1_cand.json > logs/compare_part1.log 2>&1; echo "== compare exit=$?"
for d in full envelopes stdout stderr; do echo "diff -rq $d: $(diff -rq part1_base/$d part1_cand/$d 2>&1 | wc -l) differing; files $(ls part1_cand/$d 2>/dev/null | wc -l)"; done
echo PART1-DONE $(date -u +%FT%TZ)
